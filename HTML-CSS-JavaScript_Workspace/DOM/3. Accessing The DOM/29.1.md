Of course. Here is a super simple demonstration of those five DOM access methods in a single HTML file.

---

### `index.html`

HTML

```HTML
<!DOCTYPE html>
<html lang="en">
<head>
    <title>DOM Access Demo</title>
</head>
<body>

    <h1 id="main-title">My List</h1>

    <ul>
        <li class="list-item">Apple 🍎</li>
        <li class="list-item">Orange 🍊</li>
        <li>Banana 🍌</li>
    </ul>

    <script>
        // 1. getElementById: Finds one element by its unique ID.
        const title = document.getElementById('main-title');
        title.style.color = 'blue'; // Makes the title blue

        // 2. getElementsByClassName: Finds all elements with a class.
        // Returns a collection, so we loop through it.
        const items = document.getElementsByClassName('list-item');
        for (let i = 0; i < items.length; i++) {
            items[i].style.fontWeight = 'bold'; // Makes Apple and Orange bold
        }

        // 3. getElementsByTagName: Finds all elements with a tag name.
        // Returns a collection, so we loop through it.
        const allListItems = document.getElementsByTagName('li');
        for (let i = 0; i < allListItems.length; i++) {
            allListItems[i].style.textDecoration = 'underline'; // Underlines all three items
        }

        // 4. querySelector: Finds the FIRST element matching a CSS selector.
        const firstListItem = document.querySelector('.list-item');
        firstListItem.style.backgroundColor = 'yellow'; // Highlights Apple

        // 5. querySelectorAll: Finds ALL elements matching a CSS selector.
        // Returns a collection, so we loop through it.
        const allItemsQuery = document.querySelectorAll('.list-item');
        allItemsQuery.forEach(item => {
            item.textContent += ' (Fruit)'; // Adds text to Apple and Orange
        });

    </script>

</body>
</html>
```
