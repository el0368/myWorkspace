### `app.js`

This file contains all the JavaScript logic for DOM manipulation.

JavaScript

```JavaScript
// --- 1. FINDING THINGS DEMONSTRATION ---
const findButton = document.getElementById('btn-find');

findButton.addEventListener('click', () => {
    console.log("--- Running Find/Query Methods ---");

    // a. Using document.getElementById()
    const uniqueEl = document.getElementById('unique-element');
    uniqueEl.style.border = '2px solid green'; // Visual confirmation
    console.log('Found with getElementById:', uniqueEl);

    // b. Using document.querySelector()
    // Finds the FIRST element that matches the CSS selector '.common-item'
    const firstCommonEl = document.querySelector('.common-item');
    firstCommonEl.style.border = '2px solid orange';
    console.log('Found with querySelector:', firstCommonEl);

    // c. Using document.querySelectorAll()
    // Finds ALL elements with the class 'common-item' and returns a NodeList
    const allCommonEls = document.querySelectorAll('.common-item');
    console.log('Found with querySelectorAll:', allCommonEls);
    // We can loop through the NodeList to affect each element
    allCommonEls.forEach(el => {
        el.classList.add('highlight-query');
    });

    alert('Query methods have been run! Check the console and the highlighted elements.');
});


// --- 2. CHANGING THINGS DEMONSTRATION ---
const changeButton = document.getElementById('btn-change');

changeButton.addEventListener('click', () => {
    console.log("--- Running Change/Manipulation Methods ---");
    const elementToChange = document.getElementById('change-me');

    // a. Using .textContent
    // This changes only the raw text.
    elementToChange.textContent = "My text content was changed safely. Any <strong>tags</strong> here would appear as plain text.";

    // b. Using .innerHTML
    // This parses the string as HTML.
    // Let's change a different element to show this safely.
    const container = changeButton.parentElement; // Grabs the parent container div
    container.innerHTML += '<div class="item">This was added via <strong>innerHTML</strong>.</div>';
    
    // c. Using .style
    // Note: camelCase for property names like 'backgroundColor'
    elementToChange.style.color = 'white';
    elementToChange.style.backgroundColor = 'darkslateblue';
    elementToChange.style.borderRadius = '5px';

    // d. Using .classList
    // Toggles the 'highlight-class' on and off with each click.
    elementToChange.classList.toggle('highlight-class');

    console.log("The element's text, HTML, styles, and classes have been manipulated.");
});


// --- 3. ADDING & REMOVING THINGS DEMONSTRATION ---
const addRemoveButton = document.getElementById('btn-add-remove');

addRemoveButton.addEventListener('click', () => {
    console.log("--- Running Add/Remove Methods ---");
    const list = document.getElementById('item-list');

    // a. Using document.createElement()
    // Create a new 'li' element in memory (not yet on the page)
    const newItem = document.createElement('li');
    newItem.textContent = `New item added at ${new Date().toLocaleTimeString()}`;
    newItem.classList.add('item');

    // b. Using .appendChild()
    // Attach the newly created element to the list
    list.appendChild(newItem);
    console.log("A new <li> element was created and appended.");

    // c. Using .remove()
    // Find the element we want to remove and delete it from the DOM
    const itemToRemove = document.getElementById('item-to-remove');
    if (itemToRemove) {
        itemToRemove.remove();
        console.log("The element with ID 'item-to-remove' has been removed.");
    } else {
        console.log("The 'item-to-remove' was already removed on a previous click.");
    }
});
```
