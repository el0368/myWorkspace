defmodule ElixirConcepts do
  @moduledoc """
  Demonstrates Immutability, Pure Functions, and Data Transformation.
  """

  # ==========================================
  # 1. IMMUTABILITY
  # ==========================================
  def demonstrate_immutability do
    original_list = [1, 2, 3]
    
    # In Elixir, we don't "change" the list.
    # We create a new one. 
    new_list = [0 | original_list] 

    # original_list is still [1, 2, 3]!
    # new_list is [0, 1, 2, 3]
    {original_list, new_list}
  end

  # ==========================================
  # 2. PURE FUNCTIONS
  # ==========================================
  # This is a pure function. 
  # 1. Output depends ONLY on input.
  # 2. No side effects (doesn't change global state, doesn't print to console).
  def calculate_total_price(price, tax_rate) do
    price + (price * tax_rate)
  end

  # ==========================================
  # 3. TRANSFORMATION OVER MODIFICATION
  # ==========================================
  def transform_users do
    users = [
      %{name: "Alice", age: 25, active: true},
      %{name: "Bob", age: 30, active: false},
      %{name: "Charlie", age: 35, active: true}
    ]

    # Use Enum.map to transform. The original 'users' list remains untouched.
    # Here we are "birthday-ing" every active user.
    users
    |> Enum.filter(fn user -> user.active end)
    |> Enum.map(fn user -> %{user | age: user.age + 1} end)
  end

  # ==========================================
  # 4. DEEP UPDATES (update_in)
  # ==========================================
  def update_nested_data do
    report = %{
      company: "Elixir Corp",
      metadata: %{
        location: %{
          city: "Bangkok",
          office: "Main HQ"
        }
      }
    }

    # Instead of manual nesting, we use update_in to create a new copy
    # with one specific change deep inside.
    update_in(report, [:metadata, :location, :office], fn _current -> "Remote Office" end)
  end
end

# --- Execution / Demo ---

# 1. Immutability
{old, new} = ElixirConcepts.demonstrate_immutability()
IO.puts("--- Immutability ---")
IO.inspect(old, label: "Original remains untouched")
IO.inspect(new, label: "New list created")

# 2. Pure Functions
IO.puts("\n--- Pure Function ---")
total = ElixirConcepts.calculate_total_price(100, 0.07)
IO.puts("Total for 100 at 7% tax: #{total}")

# 3. Transformation
IO.puts("\n--- Transformation (Enum.map) ---")
transformed = ElixirConcepts.transform_users()
IO.inspect(transformed, label: "Filtered and Age-Updated Users")

# 4. Deep Updates
IO.puts("\n--- Deep Updates (update_in) ---")
new_report = ElixirConcepts.update_nested_data()
IO.inspect(new_report.metadata.location.office, label: "New Office Value")
