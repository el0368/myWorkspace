#[derive(Debug)]
struct CircularLinkedList {
    nodes: Vec<Node>,
    head_index: Option<usize>,
}

#[derive(Debug)]
struct Node {
    value: i32,
    next_index: Option<usize>,
}

impl CircularLinkedList {
    fn new() -> Self {
        CircularLinkedList { 
            nodes: Vec::new(),
            head_index: None 
        }
    }
    
    fn add_node(&mut self, value: i32) {
        let new_index = self.nodes.len();
        
        if self.head_index.is_none() {
            self.nodes.push(Node { value, next_index: Some(new_index) });
            self.head_index = Some(new_index);
        } else {
            let head_idx = self.head_index.unwrap();
            let mut tail_idx = head_idx;
            
            while self.nodes[tail_idx].next_index.unwrap() != head_idx {
                tail_idx = self.nodes[tail_idx].next_index.unwrap();
            }
            
            self.nodes.push(Node { value, next_index: Some(head_idx) });
            self.nodes[tail_idx].next_index = Some(new_index);
        }
    }
    
    fn print(&self) {
        print!("Circular Linked List: ");
        if let Some(head_idx) = self.head_index {
            let mut current_idx = head_idx;
            for _ in 0..3 {
                print!("{} -> ", self.nodes[current_idx].value);
                current_idx = self.nodes[current_idx].next_index.unwrap();
            }
            println!("... (loops back to {})", self.nodes[head_idx].value);
        } else {
            println!("Empty");
        }
    }
}

fn main() {
    let mut list = CircularLinkedList::new();
    list.add_node(1);
    list.add_node(2);
    list.add_node(3);
    list.print();
}
