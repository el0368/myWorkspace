```rust
fn main() {
    // Immutable reference example
    let data = String::from("Hello, Rust!");
    let ref1 = &data; // Immutable borrow
    let ref2 = &data; // Multiple immutable borrows are allowed
    println!("Immutable ref1: {}", ref1); // Outputs: Hello, Rust!
    println!("Immutable ref2: {}", ref2); // Outputs: Hello, Rust!
    println!("Original data: {}", data); // Original data still accessible

    // Mutable reference example
    let mut mutable_data = String::from("Hello");
    let mut_ref = &mut mutable_data; // Single mutable borrow
    mut_ref.push_str(", World!");
    println!("Modified data: {}", mutable_data); // Outputs: Hello, World!
    // let mut_ref2 = &mut mutable_data; // ERROR: Cannot have multiple mutable borrows

    // Sequential mutable references (non-overlapping lifetimes)
    let mut number = 42;
    {
        let mut_ref1 = &mut number;
        *mut_ref1 += 10;
        println!("Mutated number (first ref): {}", mut_ref1); // Outputs: 52
    } // mut_ref1 lifetime ends here
    let mut_ref2 = &mut number;
    *mut_ref2 += 5;
    println!("Mutated number (second ref): {}", mut_ref2); // Outputs: 57

    // Immutable reference implements Copy
    let ref3 = ref1; // Copies the immutable reference
    println!("Copied immutable ref3: {}", ref3); // Outputs: Hello, Rust!

    // Mutable reference does not implement Copy
    let mut mutable_data2 = String::from("Test");
    let mut_ref3 = &mut mutable_data2;
    // let mut_ref4 = mut_ref3; // ERROR: Ownership moves, no copy
    let mut_ref4 = mut_ref3; // Ownership moves to mut_ref4
    mut_ref4.push_str("ing");
    println!("Moved mutable ref: {}", mut_ref4); // Outputs: Testing
    // println!("{}", mut_ref3); // ERROR: mut_ref3 moved

    // Avoiding dangling references
    let safe_ref = create_safe_reference();
    println!("Safe reference: {}", safe_ref); // Outputs: Safe data

    // Ownership with composite types
    let mut array = [String::from("One"), String::from("Two")];
    let element = &array[0]; // Borrow instead of move
    println!("Borrowed element: {}", element); // Outputs: One
    // let moved_element = array[0]; // ERROR: Would move ownership, array loses element
    let cloned_element = array[0].clone(); // Explicit clone to avoid moving
    println!("Cloned element: {}", cloned_element); // Outputs: One
    println!("Array still intact: {:?}", array); // Outputs: ["One", "Two"]

    // Composite type with Copy trait
    let numbers = [1, 2, 3];
    let num_element = numbers[0]; // Copy, since i32 implements Copy
    println!("Copied number: {}", num_element); // Outputs: 1
    println!("Numbers array: {:?}", numbers); // Outputs: [1, 2, 3]
}

// Function to demonstrate avoiding dangling references
fn create_safe_reference() -> &'static str {
    let data = "Safe data";
    data // Returns a static string slice, safe lifetime
}
```
