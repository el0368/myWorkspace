### 1. String Slices (`&str`)


```Rust
fn main() {
    let my_string = String::from("Hello, Rustaceans! 👋"); // Owned String (on Heap)
    let my_literal: &str = "Fixed text slice";            // String Literal (&str, in Program Binary)

    // --- Basic Slicing Syntax (targeting bytes) ---

    // Slice from index 0 up to (exclusive) index 5
    let slice1 = &my_string[0..5]; // Type: &str. Data points to "Hello" on Heap.
    println!("Slice 1: {}", slice1); // Output: Hello

    // Slice from index 7 up to (exclusive) index 17
    let slice2 = &my_string[7..17]; // Type: &str. Data points to "Rustaceans" on Heap.
    println!("Slice 2: {}", slice2); // Output: Rustaceans

    // Slice from the beginning (index 0) up to (exclusive) index 5
    let slice3 = &my_string[..5]; // Type: &str. Data points to "Hello" on Heap.
    println!("Slice 3: {}", slice3); // Output: Hello

    // Slice from index 7 to the end
    let slice4 = &my_string[7..]; // Type: &str. Data points to "Rustaceans! 👋" on Heap.
    println!("Slice 4: {}", slice4); // Output: Rustaceans! 👋

    // Slice the entire String (equivalent to `&my_string[0..my_string.len()]`)
    let slice5 = &my_string[..]; // Type: &str. Data points to "Hello, Rustaceans! 👋" on Heap.
    println!("Slice 5: {}", slice5); // Output: Hello, Rustaceans! 👋

    // Slicing a string literal (results in another &str)
    let literal_slice = &my_literal[..5]; // Type: &str. Data points to "Fixed" in Program Binary.
    println!("Literal Slice: {}", literal_slice); // Output: Fixed

    // --- String Slice Length: Bytes vs. Characters ---

    let unicode_string = String::from("Rustacean 👋"); // The emoji is 4 bytes in UTF-8
    let unicode_slice: &str = &unicode_string[9..]; // Slice starting from 'space' + emoji
                                                     // " 👋" from original string

    println!("\nUnicode Slice: '{}'", unicode_slice); // Output:  👋
    println!("Bytes in slice: {}", unicode_slice.len()); // Output: 5 (1 for space, 4 for emoji)
    println!("Chars in slice: {}", unicode_slice.chars().count()); // Output: 2 (1 for space, 1 for emoji)

    // Example of invalid byte slicing (would panic if `get` wasn't used)
    // let invalid_slice = &unicode_string[10..11]; // This would attempt to slice *mid-character*
    // println!("Invalid slice (byte-level): {}", invalid_slice); // Panics at runtime! (or compile error with `get`)
}
```

### 2. Array Slices (`&[T]`)


```Rust
fn main() {
    let my_array: [i32; 5] = [10, 20, 30, 40, 50]; // Fixed-size array (on Stack)

    // --- Basic Slicing Syntax (targeting elements by index) ---

    // Slice from index 0 up to (exclusive) index 3
    let array_slice1 = &my_array[0..3]; // Type: &[i32]. Data points to [10, 20, 30] on Stack.
    println!("Array Slice 1: {:?}", array_slice1); // Output: [10, 20, 30]

    // Slice from index 2 to the end
    let array_slice2 = &my_array[2..]; // Type: &[i32]. Data points to [30, 40, 50] on Stack.
    println!("Array Slice 2: {:?}", array_slice2); // Output: [30, 40, 50]

    // Slice the entire array
    let array_slice3 = &my_array[..]; // Type: &[i32]. Data points to [10, 20, 30, 40, 50] on Stack.
    println!("Array Slice 3: {:?}", array_slice3); // Output: [10, 20, 30, 40, 50]

    // --- Inclusive Range for Slices (`..=`) ---
    let array_slice_inclusive = &my_array[0..=2]; // Type: &[i32]. Data points to [10, 20, 30] on Stack.
    println!("Array Slice Inclusive: {:?}", array_slice_inclusive); // Output: [10, 20, 30]
}
```

### 3. Deref Coercion with Strings (`&String` to `&str`)


```Rust
// Function expects an &str parameter
fn print_string_slice(s: &str) {
    println!("Function received &str: {}", s);
}

// Function expects an &String parameter (less common for read-only)
fn print_string_ref(s: &String) {
    println!("Function received &String: {}", s);
}

fn main() {
    let owned_string = String::from("Deref Coercion"); // `owned_string` (String) on Stack, data on Heap.
    let string_literal: &str = "Literal Text";         // `string_literal` (&str) on Stack, data in Program Binary.

    println!("\n--- Testing &str Parameter ---");
    print_string_slice(string_literal); // Direct match: &str -> &str
    print_string_slice(&owned_string);  // Deref Coercion: &String -> &str (Automatic, works!)
    // This is because `String` implements `Deref<Target = str>`, allowing `&String` to convert to `&str`.

    println!("\n--- Testing &String Parameter ---");
    print_string_ref(&owned_string); // Direct match: &String -> &String

    // This will NOT compile. Deref Coercion does not apply in reverse.
    // Rust cannot guarantee that an arbitrary `&str` comes from an `&String`.
    // print_string_ref(string_literal); // ERROR: expected `&String`, found `&str`
}
```

### 4. Deref Coercion with Arrays (`&[T; N]` to `&[T]`)


```Rust
// Function expects an array slice (dynamic length)
fn process_array_slice(slice: &[i32]) { // `slice` is a &[i32] (pointer + length) on Stack
    println!("Processing slice of length {}: {:?}", slice.len(), slice);
}

// Function expects a fixed-size array reference
fn process_fixed_array(array_ref: &[i32; 3]) { // `array_ref` is a &[i32; 3] (pointer) on Stack
    println!("Processing fixed array of length {}: {:?}", array_ref.len(), array_ref);
}

fn main() {
    let my_fixed_array: [i32; 3] = [100, 200, 300]; // `my_fixed_array` ([i32; 3]) on Stack
    let my_dynamic_slice: &[i32] = &[1, 2, 3, 4, 5]; // `my_dynamic_slice` (&[i32]) on Stack, points to anonymous array data on Stack/Binary.

    println!("\n--- Testing &[i32] Parameter ---");
    process_array_slice(&my_fixed_array);  // Deref Coercion: &[i32; 3] -> &[i32] (Automatic, works!)
    process_array_slice(my_dynamic_slice); // Direct match: &[i32] -> &[i32]

    println!("\n--- Testing &[i32; 3] Parameter ---");
    process_fixed_array(&my_fixed_array); // Direct match: &[i32; 3] -> &[i32; 3]

    // This will NOT compile. Deref Coercion does not apply in reverse.
    // Rust cannot convert a generic &[i32] to a specific &[i32; 3] because the length isn't guaranteed at compile time.
    // process_fixed_array(my_dynamic_slice); // ERROR: expected `&[i32; 3]`, found `&[i32]`
}
```
