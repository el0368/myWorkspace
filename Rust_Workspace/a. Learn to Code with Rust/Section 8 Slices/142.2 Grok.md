fn main() {
    // String slice example
    let text = String::from("Hello, Rust!");
    let slice1: &str = &text[0..5]; // Slice from byte 0 to 5 (exclusive)
    let slice2: &str = &text[7..]; // Slice from byte 7 to end
    let slice3: &str = &text[..]; // Slice of entire string
    println!("Slice 1: {}", slice1); // Outputs: Hello
    println!("Slice 2: {}", slice2); // Outputs: Rust!
    println!("Slice 3: {}", slice3); // Outputs: Hello, Rust!

    // String literal is a &str
    let literal: &str = "World";
    println!("String literal slice: {}", literal); // Outputs: World

    // Array slice example
    let numbers = [1, 2, 3, 4, 5];
    let array_slice1: &[i32] = &numbers[1..4]; // Slice from index 1 to 4 (exclusive)
    let array_slice2: &[i32] = &numbers[..3]; // Slice from start to index 3
    let array_slice3: &[i32] = &numbers[..]; // Slice of entire array
    println!("Array slice 1: {:?}", array_slice1); // Outputs: [2, 3, 4]
    println!("Array slice 2: {:?}", array_slice2); // Outputs: [1, 2, 3]
    println!("Array slice 3: {:?}", array_slice3); // Outputs: [1, 2, 3, 4, 5]

    // Inclusive range syntax
    let inclusive_slice: &[i32] = &numbers[1..=3]; // Slice from index 1 to 3 (inclusive)
    println!("Inclusive array slice: {:?}", inclusive_slice); // Outputs: [2, 3, 4]

    // Length of a string slice (byte count)
    let emoji = "😊 Rust";
    let emoji_slice: &str = &emoji[..]; // Entire string
    println!("Length of emoji slice: {}", emoji_slice.len()); // Outputs: 8 (bytes, not characters)

    // Deref coercion example
    let string = String::from("Deref Coercion");
    let str_slice: &str = &string; // Deref coercion from &String to &str
    print_slice(str_slice); // Outputs: Slice: Deref Coercion
    print_slice(&string); // Outputs: Slice: Deref Coercion (coerced from &String to &str)

    // Array deref coercion
    let array_ref: &[i32; 5] = &numbers;
    print_array_slice(array_ref); // Outputs: Array slice: [1, 2, 3, 4, 5]
    print_array_slice(&numbers[..]); // Outputs: Array slice: [1, 2, 3, 4, 5]
}

// Function accepting a string slice (&str) for flexibility
fn print_slice(slice: &str) {
    println!("Slice: {}", slice);
}

// Function accepting an array slice (&[i32]) for flexibility
fn print_array_slice(slice: &[i32]) {
    println!("Array slice: {:?}", slice);
}
