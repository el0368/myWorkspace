```Rust
use std::collections::HashMap;

fn main() {
    // Creating a HashMap with explicit type annotation
    let mut coffee_pairings: HashMap<&str, &str> = HashMap::new();
    
    // Inserting key-value pairs
    coffee_pairings.insert("Latte", "Oat Milk");
    coffee_pairings.insert("Flat White", "Almond Milk");
    println!("Initial HashMap: {:?}", coffee_pairings); // Outputs: {"Latte": "Oat Milk", "Flat White": "Almond Milk"}
    
    // Length of HashMap (counts key-value pairs)
    println!("Length: {}", coffee_pairings.len()); // Outputs: 2
    
    // Insert with existing key (overwrites value)
    coffee_pairings.insert("Latte", "Pistachio Milk");
    println!("After insert (overwrite): {:?}", coffee_pairings); // Outputs: {"Latte": "Pistachio Milk", "Flat White": "Almond Milk"}
    
    // Remove a key-value pair
    let removed = coffee_pairings.remove("Flat White");
    println!("Removed value: {:?}", removed); // Outputs: Some("Almond Milk")
    println!("After remove: {:?}", coffee_pairings); // Outputs: {"Latte": "Pistachio Milk"}
    
    // Remove non-existent key
    let removed = coffee_pairings.remove("Cappuccino");
    println!("Removed non-existent: {:?}", removed); // Outputs: None
    
    // Using entry and or_insert (conditional insert)
    coffee_pairings.entry("Latte").or_insert("Soy Milk"); // Does nothing, key exists
    coffee_pairings.entry("Cappuccino").or_insert("Soy Milk"); // Adds new pair
    println!("After entry/or_insert: {:?}", coffee_pairings); // Outputs: {"Latte": "Pistachio Milk", "Cappuccino": "Soy Milk"}
    
    // Accessing values with square brackets (panics if key doesn't exist)
    let value = coffee_pairings["Latte"];
    println!("Value for Latte: {}", value); // Outputs: Pistachio Milk
    // let value = coffee_pairings["Espresso"]; // Would panic
    
    // Accessing values with get (safe, returns Option)
    let value = coffee_pairings.get("Latte");
    println!("Get Latte: {:?}", value); // Outputs: Some("Pistachio Milk")
    let value = coffee_pairings.get("Espresso");
    println!("Get Espresso: {:?}", value); // Outputs: None
    
    // Using copied and unwrap_or for safe value extraction
    let value = coffee_pairings.get("Latte").copied().unwrap_or("Unknown Milk");
    println!("Safe access Latte: {}", value); // Outputs: Pistachio Milk
    let value = coffee_pairings.get("Espresso").copied().unwrap_or("Unknown Milk");
    println!("Safe access Espresso: {}", value); // Outputs: Unknown Milk
}
```
