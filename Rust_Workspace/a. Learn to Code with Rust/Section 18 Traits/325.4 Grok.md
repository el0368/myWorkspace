// Comprehensive example demonstrating traits in Rust, covering all details from the transcript

use std::fmt::{self, Formatter};

// Custom trait definition
trait Animal {
    // Required method signature
    fn make_sound(&self) -> &'static str;

    // Method with default implementation
    fn describe(&self) -> &'static str {
        "This is an animal"
    }

    // Getter method (to access data indirectly)
    fn get_name(&self) -> &str;

    // Setter method (to set data indirectly)
    fn set_name(&mut self, name: String);

    // Associated constant
    const LEGS: u8 = 4;

    // Associated type (placeholder for future type)
    type Food;
    fn favorite_food(&self) -> Self::Food;
}

// Super trait (parent trait)
trait LivingBeing {
    fn breathe(&self) -> &'static str;
}

// Subtrait (child trait) inheriting from super trait
trait Mammal: LivingBeing {
    fn give_birth(&self) -> &'static str;
}

// Struct implementing traits
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
struct Dog {
    name: String,
    age: u8,
}

impl Animal for Dog {
    fn make_sound(&self) -> &'static str {
        "Woof!"
    }

    // Overwriting default method
    fn describe(&self) -> &'static str {
        "This is a dog"
    }

    fn get_name(&self) -> &str {
        &self.name
    }

    fn set_name(&mut self, name: String) {
        self.name = name;
    }

    // Overwriting associated constant
    const LEGS: u8 = 4;

    // Implementing associated type
    type Food = &'static str;
    fn favorite_food(&self) -> Self::Food {
        "Bone"
    }
}

impl LivingBeing for Dog {
    fn breathe(&self) -> &'static str {
        "Inhale... Exhale..."
    }
}

impl Mammal for Dog {
    fn give_birth(&self) -> &'static str {
        "Live birth"
    }
}

// Separate impl block for custom methods (not tied to a trait)
impl Dog {
    fn new(name: String, age: u8) -> Self {
        Dog { name, age }
    }
}

// Implementing Display trait
impl fmt::Display for Dog {
    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
        write!(f, "Dog named {} aged {}", self.name, self.age)
    }
}

// Implementing Drop trait
impl Drop for Dog {
    fn drop(&mut self) {
        println!("Dropping Dog: {}", self.name);
    }
}

// Another struct implementing Animal
#[derive(Debug, Clone, PartialEq, PartialOrd)]
struct Cat {
    name: String,
}

impl Animal for Cat {
    fn make_sound(&self) -> &'static str {
        "Meow!"
    }

    fn get_name(&self) -> &str {
        &self.name
    }

    fn set_name(&mut self, name: String) {
        self.name = name;
    }

    // Overwriting associated constant
    const LEGS: u8 = 4;

    type Food = &'static str;
    fn favorite_food(&self) -> Self::Food {
        "Fish"
    }
}

impl LivingBeing for Cat {
    fn breathe(&self) -> &'static str {
        "Inhale... Exhale..."
    }
}

impl Mammal for Cat {
    fn give_birth(&self) -> &'static str {
        "Live birth"
    }
}

// Function with trait bounds (option 1: impl Trait)
fn print_sound(animal: impl Animal) {
    println!("Sound: {}", animal.make_sound());
}

// Function with trait bounds (option 2: generics with : Trait)
fn print_description<T: Animal>(animal: T) {
    println!("Description: {}", animal.describe());
}

// Function with multiple trait bounds
fn print_info<T: Animal + Display>(animal: T) {
    println!("Info: {}", animal);
}

// Function with where clause for trait bounds
fn print_favorite_food<T>(animal: T)
where
    T: Animal,
{
    println!("Favorite food: {}", animal.favorite_food());
}

// Demonstrating NaN with PartialEq/PartialOrd
fn nan_example() {
    let nan1 = f64::NAN;
    let nan2 = f64::NAN;
    println!("NaN == NaN: {}", nan1 == nan2); // Outputs: false (PartialEq allows this)
    // But f64 does not implement Eq, so cannot use eq() directly for strict equality
}

fn main() {
    // Creating instances
    let mut dog = Dog::new(String::from("Buddy"), 3);
    let cat = Cat { name: String::from("Whiskers") };

    // Invoking trait methods
    println!("Dog sound: {}", dog.make_sound()); // Outputs: Woof!
    println!("Cat sound: {}", cat.make_sound()); // Outputs: Meow!

    // Default method
    println!("Dog describe: {}", dog.describe()); // Outputs: This is a dog

    // Getter and setter
    println!("Dog name: {}", dog.get_name()); // Outputs: Buddy
    dog.set_name(String::from("Max"));
    println!("New dog name: {}", dog.get_name()); // Outputs: Max

    // Associated constant
    println!("Dog legs: {}", Dog::LEGS); // Outputs: 4

    // Associated type
    println!("Dog favorite food: {}", dog.favorite_food()); // Outputs: Bone

    // Super/sub trait
    println!("Dog breathe: {}", dog.breathe()); // Outputs: Inhale... Exhale...
    println!("Dog give birth: {}", dog.give_birth()); // Outputs: Live birth

    // Display trait
    println!("Dog display: {}", dog); // Outputs: Dog named Max aged 3

    // Drop trait (automatically called at end of scope)
    {
        let temp_dog = Dog::new(String::from("Temp"), 1);
    } // Outputs: Dropping Dog: Temp

    // Clone trait
    let dog_clone = dog.clone();
    println!("Dog clone: {:?}", dog_clone); // Outputs: Dog { name: "Max", age: 3 }

    // Copy trait (since derived, works implicitly)
    let dog_copy = dog; // Copies
    println!("Dog copy: {:?}", dog_copy); // Outputs: Dog { name: "Max", age: 3 }

    // PartialEq and Eq
    println!("Dog == Dog copy: {}", dog == dog_copy); // Outputs: true

    // PartialOrd and Ord
    println!("Dog < Dog copy: {}", dog < dog_copy); // Outputs: false (based on derived ord)

    // Trait bounds examples
    print_sound(&dog); // Outputs: Sound: Woof!
    print_description(&cat); // Outputs: Description: This is an animal
    print_info(dog_clone); // Outputs: Info: Dog named Max aged 3
    print_favorite_food(&cat); // Outputs: Favorite food: Fish

    // Trait objects (dynamic dispatch)
    let animals: Vec<&dyn Animal> = vec![&dog, &cat]; // Different types in vector
    for animal in animals {
        println!("Trait object sound: {}", animal.make_sound());
        // Outputs: Woof! then Meow!
    }

    // NaN example
    nan_example(); // Outputs: NaN == NaN: false
}
