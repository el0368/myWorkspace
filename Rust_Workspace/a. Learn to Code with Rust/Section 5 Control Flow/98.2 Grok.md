```rust
fn main() {
    // If statement example
    let number = 42;
    let result = if number > 50 {
        "Large"
    } else if number > 20 {
        "Medium"
    } else {
        "Small"
    };
    println!("Number size: {}", result); // Outputs: Medium

    // Match statement example
    let value = true;
    let match_result = match value {
        true => "It's true!",
        false => "It's false!",
    };
    println!("Match result: {}", match_result); // Outputs: It's true!

    // Match with numbers and catch-all
    let num = 7;
    let num_description = match num {
        1 => "One",
        2 => "Two",
        3 => "Three",
        _ => "Other",
    };
    println!("Number description: {}", num_description); // Outputs: Other

    // Loop example with break and continue
    let mut counter = 0;
    loop {
        counter += 1;
        if counter == 3 {
            continue; // Skip this iteration
        }
        if counter > 5 {
            break; // Exit the loop
        }
        println!("Loop iteration: {}", counter); // Outputs: 1, 2, 4, 5
    }

    // While loop example
    let mut count = 0;
    while count < 5 {
        count += 1;
        println!("While iteration: {}", count); // Outputs: 1, 2, 3, 4, 5
    }

    // Recursion example
    let factorial_result = factorial(5);
    println!("Factorial of 5: {}", factorial_result); // Outputs: 120
}

// Recursive function with base case
fn factorial(n: u32) -> u32 {
    if n <= 1 {
        1 // Base case
    } else {
        n * factorial(n - 1) // Recursive case
    }
}
```
