// --- LinkedList and Node Definitions ---
#[derive(Debug)]
struct Node {
    value: i32,
    next: Option<Box<Node>>,
}

#[derive(Debug)]
struct LinkedList {
    head: Option<Box<Node>>,
}

// --- Implementation Block for LinkedList ---
impl LinkedList {
    // Associated function (constructor) using Self alias
    fn new() -> Self {
        LinkedList { head: None }
    }

    // Method with mutable reference (&mut self) to push a new node
    fn push(&mut self, value: i32) {
        let new_node = Box::new(Node {
            value,
            next: self.head.take(),
        });
        self.head = Some(new_node);
    }

    // Method with immutable reference (&self) to display the list
    fn display_list(&self) {
        let mut current = &self.head;
        while let Some(node) = current {
            print!("{} -> ", node.value);
            current = &node.next;
        }
        println!("None");
    }

    // Method with ownership (self) to consume and clear the list (example of consumption)
    fn clear_list(self) {
        println!("Clearing list (consuming ownership)...");
        // Since self owns the list, it will be deallocated when this method ends
    }
}
