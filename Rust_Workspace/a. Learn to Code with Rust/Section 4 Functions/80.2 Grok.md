```rust
// Example functions demonstrating concepts from the transcript

// Function with parameters and implicit return
fn add_numbers(a: i32, b: i32) -> i32 {
    a + b // Implicit return (no semicolon)
}

// Function with explicit return
fn subtract_numbers(x: i32, y: i32) -> i32 {
    return x - y; // Explicit return
}

// Function returning unit (no return type specified)
fn print_hello() {
    println!("Hello from a function!"); // Returns unit ()
}

// Function using a block to compute a value
fn compute_with_block(num: i32) -> i32 {
    let result = {
        let temp = num * 2;
        temp + 1 // Block produces this value
    }; // Semicolon here since assigning to variable
    result
}

fn main() {
    // Invoking functions with arguments
    let sum = add_numbers(5, 3);
    println!("Sum: {}", sum); // Outputs: 8

    let difference = subtract_numbers(10, 4);
    println!("Difference: {}", difference); // Outputs: 6

    print_hello(); // Outputs: Hello from a function!

    let block_result = compute_with_block(7);
    println!("Block result: {}", block_result); // Outputs: 15
}
```
