```rust
// Example demonstrating enums, their variants, match, if let, and methods in Rust

// Enum definition with different variant types
enum Message {
    // Empty variant
    Quit,
    // Tuple variant
    Move(i32, i32),
    // Struct variant
    Write { text: String, priority: u8 },
    // Tuple variant with mixed types
    Color(u8, u8, u8),
}

impl Message {
    // Method with immutable reference
    fn describe(&self) -> &'static str {
        match self {
            Message::Quit => "Quit message",
            Message::Move(..) => "Move message",
            Message::Write { .. } => "Write message",
            Message::Color(..) => "Color message",
        }
    }

    // Method with mutable reference
    fn update_priority(&mut self) {
        if let Message::Write { priority, .. } = self {
            *priority += 1;
        }
    }
}

fn main() {
    // Creating enum instances
    let quit_msg = Message::Quit;
    let move_msg = Message::Move(10, 20);
    let write_msg = Message::Write {
        text: String::from("Hello"),
        priority: 5,
    };
    let color_msg = Message::Color(255, 128, 0);

    // Ownership: Enums don't implement Copy by default
    let msg1 = Message::Write {
        text: String::from("Test"),
        priority: 1,
    };
    let msg2 = msg1; // Ownership moves to msg2
    // println!("{:?}", msg1); // ERROR: msg1 moved
    println!("msg2: {:?}", msg2); // Outputs: Write { text: "Test", priority: 1 }

    // Match statement to handle all variants
    let dynamic_msg = Message::Move(5, 7);
    let description = match dynamic_msg {
        Message::Quit => "Received a quit message",
        Message::Move(x, y) => {
            // Extract tuple variant data
            format!("Moving to ({}, {})", x, y)
        }
        Message::Write { text, priority } => {
            // Extract struct variant data
            format!("Writing '{}' with priority {}", text, priority)
        }
        Message::Color(r, g, b) => {
            format!("Color RGB({}, {}, {})", r, g, b)
        }
    };
    println!("Match result: {}", description); // Outputs: Moving to (5, 7)

    // Match with specific value check
    let priority_msg = Message::Write {
        text: String::from("Urgent"),
        priority: 10,
    };
    let priority_check = match priority_msg {
        Message::Write { priority: 10, .. } => "High priority message",
        Message::Write { .. } => "Other priority message",
        _ => "Not a write message",
    };
    println!("Priority check: {}", priority_check); // Outputs: High priority message

    // If let construct for single variant
    let some_msg = Message::Write {
        text: String::from("Note"),
        priority: 3,
    };
    if let Message::Write { text, priority } = &some_msg {
        println!("If let extracted: text='{}', priority={}", text, priority); // Outputs: text='Note', priority=3
    } else {
        println!("Not a write message");
    }

    // Method calls
    println!("Quit description: {}", quit_msg.describe()); // Outputs: Quit message
    println!("Move description: {}", move_msg.describe()); // Outputs: Move message

    // Mutable method call
    let mut mutable_msg = Message::Write {
        text: String::from("Changeable"),
        priority: 2,
    };
    mutable_msg.update_priority();
    if let Message::Write { priority, .. } = mutable_msg {
        println!("Updated priority: {}", priority); // Outputs: 3
    }
}
```
