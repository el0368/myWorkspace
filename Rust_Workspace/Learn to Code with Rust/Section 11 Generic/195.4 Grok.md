```Rust
// Example demonstrating generics in Rust

// Generic struct definition with one generic type
struct Pair<T> {
    first: T,
    second: T,
}

// Generic struct with multiple generic types
struct MixedPair<T, U> {
    first: T,
    second: U,
}

// Generic enum definition
enum Status<T> {
    Success(T),
    Failure,
}

// Generic function with one generic type
fn identity<T>(value: T) -> T {
    value
}

// Generic function with multiple generic types
fn combine<T, U>(first: T, second: U) -> (T, U) {
    (first, second)
}

// Implementation block for generic struct (works for any T)
impl<T> Pair<T> {
    fn new(first: T, second: T) -> Self {
        Pair { first, second }
    }

    fn swap(self) -> Pair<T> {
        Pair {
            first: self.second,
            second: self.first,
        }
    }
}

// Implementation block for specific type (i32)
impl Pair<i32> {
    fn sum(&self) -> i32 {
        self.first + self.second
    }
}

// Implementation for MixedPair with multiple generics
impl<T, U> MixedPair<T, U> {
    fn new(first: T, second: U) -> Self {
        MixedPair { first, second }
    }
}

fn main() {
    // Using generic function with single type
    let int_result = identity(42); // T is i32
    let str_result = identity("Hello"); // T is &str
    println!("Identity int: {}", int_result); // Outputs: 42
    println!("Identity str: {}", str_result); // Outputs: Hello

    // Using generic function with multiple types
    let combined = combine(10, "Rust"); // T is i32, U is &str
    println!("Combined: {:?}", combined); // Outputs: (10, "Rust")

    // Using generic struct with same type
    let int_pair = Pair::new(10, 20); // T is i32
    let str_pair = Pair::new("Alice", "Bob"); // T is &str
    println!("Int pair: ({}, {})", int_pair.first, int_pair.second); // Outputs: (10, 20)
    println!("Str pair: ({}, {})", str_pair.first, str_pair.second); // Outputs: (Alice, Bob)

    // Using generic struct with swap method
    let swapped_pair = int_pair.swap();
    println!("Swapped int pair: ({}, {})", swapped_pair.first, swapped_pair.second); // Outputs: (20, 10)

    // Using method specific to i32
    let int_pair = Pair::new(5, 15);
    println!("Sum of int pair: {}", int_pair.sum()); // Outputs: 20

    // Using generic struct with multiple types
    let mixed_pair = MixedPair::new(42, String::from("Rust")); // T is i32, U is String
    println!("Mixed pair: ({}, {})", mixed_pair.first, mixed_pair.second); // Outputs: (42, Rust)

    // Using generic enum
    let success: Status<i32> = Status::Success(100);
    let failure: Status<String> = Status::Failure;
    match success {
        Status::Success(value) => println!("Success value: {}", value), // Outputs: 100
        Status::Failure => println!("Failure"),
    }
    match failure {
        Status::Success(_) => println!("Success"),
        Status::Failure => println!("Failure"), // Outputs: Failure
    }
}
```
