```rust
fn main() {
    // Stack data with Copy trait
    let x = 42; // i32, implements Copy
    let y = x; // Creates a copy, x remains valid
    println!("x: {}, y: {}", x, y); // Outputs: x: 42, y: 42

    // Heap data without Copy trait
    let s1 = String::from("Hello"); // String, does not implement Copy
    let s2 = s1; // Ownership moves to s2
    // println!("s1: {}", s1); // ERROR: s1 is moved
    println!("s2: {}", s2); // Outputs: Hello

    // Using clone to create a duplicate
    let s3 = String::from("Rust");
    let s4 = s3.clone(); // Explicitly creates a copy
    println!("s3: {}, s4: {}", s3, s4); // Outputs: s3: Rust, s4: Rust

    // Ownership with function parameters
    let s5 = String::from("World");
    take_ownership(s5); // Ownership moves to function
    // println!("s5: {}", s5); // ERROR: s5 is moved

    let num = 100;
    copy_value(num); // Creates a copy, num remains valid
    println!("num after function: {}", num); // Outputs: 100

    // References (borrowing)
    let s6 = String::from("Borrow");
    let reference = &s6; // Immutable borrow
    println!("Reference: {}", reference); // Outputs: Borrow (auto-dereferenced)
    println!("Original s6: {}", s6); // Outputs: Borrow, s6 still valid

    // Dereferencing
    let value = *reference; // Explicit dereference (not needed for printing)
    println!("Dereferenced value: {}", value); // Outputs: Borrow

    // Scope and ownership
    let s7: String;
    {
        s7 = String::from("Scoped");
        println!("Inside scope: {}", s7); // Outputs: Scoped
    } // s7 goes out of scope, memory deallocated
    // println!("Outside scope: {}", s7); // ERROR: s7 out of scope

    // Avoiding dangling references
    let safe_ref = create_safe_reference();
    println!("Safe reference: {}", safe_ref); // Outputs: Safe data
}

// Function that takes ownership (moves String)
fn take_ownership(s: String) {
    println!("Took ownership: {}", s); // Outputs: World
} // s goes out of scope, memory deallocated

// Function that copies stack data
fn copy_value(n: i32) {
    println!("Copied value: {}", n); // Outputs: 100
}

// Function to return a safe reference
fn create_safe_reference() -> &'static str {
    let data = "Safe data";
    data // Returns a static string slice, safe lifetime
}
```
