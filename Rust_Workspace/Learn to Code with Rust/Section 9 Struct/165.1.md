// --- SECTION REVIEW: STRUCTS ---

// A named-field struct: a container for related data with descriptive names.
// It acts as a blueprint.
#[derive(Debug)] // The `derive` attribute automatically implements the Debug trait for printing.
struct Coffee {
    // These are the fields, with names and types.
    name: String,
    price: f64,
    is_hot: bool,
}

// A tuple-like struct: orders fields by position, not name.
struct Coordinates(f64, f64);

// A unit-like struct: a struct with no fields.
struct BrewState;

// The `impl` block defines methods and associated functions for the `Coffee` struct.
impl Coffee {
    // An associated function (constructor). It does not take `self`.
    // By convention, constructors are often named `new`.
    // The `Self` keyword is an alias for the `Coffee` struct type.
    fn new(name: String, price: f64, is_hot: bool) -> Self {
        Self { name, price, is_hot } // Field init shorthand for `name: name` etc.
    }

    // A method that takes an immutable reference `&self` (a borrow).
    // This is the most common `self` signature for read-only access.
    // It prevents ownership from moving and does not allow mutation.
    fn display_details(&self) {
        println!("Details: {} costs {}. Hot: {}", self.name, self.price, self.is_hot);
    }

    // A method that takes a mutable reference `&mut self`.
    // It is a mutable borrow, allowing the method to modify the struct instance.
    fn change_to_iced(&mut self) {
        self.is_hot = false;
        println!("Changed to iced.");
    }

    // A method that takes immutable ownership `self`.
    // This consumes the struct instance, and it cannot be used afterward.
    fn drink(self) {
        println!("Drinking the {}...", self.name);
        // The `self` parameter's ownership is dropped here.
    }
}

fn main() {
    // --- 1. Creating Struct Instances (Named-Field) ---

    // `name` variable is used to show field init shorthand.
    let name = String::from("Mocha");

    // Creating a `Coffee` instance from its blueprint.
    let coffee = Coffee {
        name: name, // Standard field initialization.
        price: 4.99,
        is_hot: true,
    };
    // The `name` variable is now moved into the `coffee` struct and is invalid.
    // println!("{}", name); // ERROR: borrow of moved value: `name`

    // --- 2. Accessing Struct Fields & Ownership ---

    // Access fields using dot notation (`.`).
    println!("\nAccessing fields:");
    println!("Coffee name is: {}", coffee.name);
    println!("Coffee price is: {}", coffee.price);

    // Creating a new variable by moving a non-Copy field (`String`).
    // `coffee.name` loses ownership of its `String` and becomes invalid.
    let my_favorite_drink = coffee.name; // Ownership of "Mocha" moves to `my_favorite_drink`.
    println!("My favorite drink: {}", my_favorite_drink);
    // println!("Original coffee name: {}", coffee.name); // ERROR: borrow of moved value
    // println!("Original coffee: {:?}", coffee); // ERROR: part of `coffee` moved

    // --- 3. Struct Update Syntax (`..`) ---

    let cappuccino = Coffee {
        name: String::from("Cappuccino"),
        ..Coffee::new(String::from("Any"), 3.50, false) // `..` copies remaining fields from this temporary `Coffee`.
                                                         // The values `3.50` and `false` are copied because they are `Copy` types.
    };

    // --- 4. Deriving and Printing Structs ---

    println!("\nDerived Debug trait output:");
    println!("Struct: {:?}", cappuccino); // Standard debug output
    println!("Pretty Struct: {:#?}", cappuccino); // Pretty debug output

    // --- 5. Calling Methods and Associated Functions ---

    // Calling a method using dot notation on an immutable instance.
    cappuccino.display_details();

    // Calling a method that mutates the instance (requires `&mut` borrow).
    let mut latte = Coffee::new(String::from("Latte"), 4.00, true);
    latte.display_details(); // `&self` is automatically passed.
    latte.change_to_iced();  // `&mut self` is automatically passed.
    latte.display_details();

    // Calling a method that takes ownership (`self`).
    // This consumes the `latte` instance.
    latte.drink();
    // println!("After drinking: {:?}", latte); // ERROR: borrow of moved value: `latte`

    // Calling an associated function on the struct type itself using `::`.
    let constructed_coffee = Coffee::new(String::from("Americano"), 3.00, false);
    constructed_coffee.display_details();

    // --- 6. Tuple-like Structs ---

    let origin = Coordinates(0.0, 0.0);
    println!("\nTuple struct: x={}, y={}", origin.0, origin.1); // Access fields by index `.0`, `.1`.

    // --- 7. Unit-like Structs ---

    let _status = BrewState; // Instantiation requires no parentheses or braces.
}
