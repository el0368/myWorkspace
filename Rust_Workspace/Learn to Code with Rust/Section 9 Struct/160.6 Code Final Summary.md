```rust
// =============================================================================
// PART 1: Initial Struct Setup (Foundation)
// =============================================================================

#[derive(Debug)]
struct Computer {
    cpu: String,
    memory: u32,
    hard_drive_capacity: u32,
}

impl Computer {
    // Associated function (constructor) to create a new instance
    fn new(cpu: String, memory: u32, hard_drive_capacity: u32) -> Self {
        // Use field init shorthand to simplify instantiation
        Self { cpu, memory, hard_drive_capacity }
    }
}

// Example usage of just the constructor:
fn part1_example() {
    println!("=== PART 1: Initial Setup ===");
    let computer = Computer::new(String::from("M3 Max"), 64, 2);
    println!("Initial computer stats: {:#?}", computer);
}

// =============================================================================
// PART 2: Non-Chainable Methods (The Problem)
// =============================================================================

// Adding methods that return () - these cannot be chained
impl Computer {
    // These methods return nothing (implicitly `()`), so they are not chainable
    fn upgrade_cpu_old(&mut self, new_cpu: String) {
        self.cpu = new_cpu;
        // returns `()` implicitly
    }
    
    fn upgrade_memory_old(&mut self, new_memory: u32) {
        self.memory = new_memory;
        // returns `()` implicitly
    }
    
    fn upgrade_hard_drive_capacity_old(&mut self, new_capacity: u32) {
        self.hard_drive_capacity = new_capacity;
        // returns `()` implicitly
    }
}

// Example of the problem - repetitive, non-chainable code:
fn part2_example() {
    println!("\n=== PART 2: Non-Chainable Methods (The Problem) ===");
    let mut computer = Computer::new(String::from("M3 Max"), 64, 2);
    
    // This is the repetitive, non-chainable code we want to improve
    computer.upgrade_cpu_old(String::from("M4 Max"));
    computer.upgrade_memory_old(128);
    computer.upgrade_hard_drive_capacity_old(4);
    
    println!("Computer after old-style upgrades: {:#?}", computer);
}

// =============================================================================
// PART 3: Refactored Methods for Chaining (The Solution)
// =============================================================================

// Adding the refactored, chainable methods
impl Computer {
    // REFACTORED METHOD: Now returns a mutable reference to itself (`&mut Self`)
    fn upgrade_cpu(&mut self, new_cpu: String) -> &mut Self {
        self.cpu = new_cpu; // Modifies the instance in place
        self // The mutable reference is returned, without a semicolon
    }
    
    // REFACTORED METHOD: Returns a mutable reference for chaining
    fn upgrade_memory(&mut self, new_memory: u32) -> &mut Self {
        self.memory = new_memory;
        self // The mutable reference is returned
    }
    
    // REFACTORED METHOD: Returns a mutable reference for chaining
    fn upgrade_hard_drive_capacity(&mut self, new_capacity: u32) -> &mut Self {
        self.hard_drive_capacity = new_capacity;
        self // The mutable reference is returned
    }
}

// =============================================================================
// PART 4 & 5: The Final Code - Chaining in Action & Verification
// =============================================================================

fn main() {
    // Run all examples to show the progression
    part1_example();
    part2_example();
    
    println!("\n=== PARTS 4 & 5: Method Chaining in Action ===");
    
    // Create the initial computer instance (must be mutable to allow changes)
    let mut computer = Computer::new(String::from("M3 Max"), 64, 2);
    println!("Before chaining: {:#?}", computer);
    
    // This is the elegant, chainable syntax of the Builder Pattern
    // Each method returns `&mut self`, allowing the next `.method()` call
    computer.upgrade_cpu(String::from("M4 Max"))
            .upgrade_memory(128)
            .upgrade_hard_drive_capacity(4);
    
    // After the entire chain has run, we can verify the results
    println!("Final computer stats: {:#?}", computer);
    
    // Additional demonstration: Multiple chains on the same object
    println!("\n=== Additional Chaining Example ===");
    computer.upgrade_cpu(String::from("M5 Pro"))
            .upgrade_memory(256);
    
    println!("After second chain: {:#?}", computer);
    
    // Demonstration: Creating and chaining in one expression
    println!("\n=== Creating and Chaining in One Expression ===");
    let mut gaming_rig = Computer::new(String::from("Intel i5"), 16, 1);
    gaming_rig.upgrade_cpu(String::from("AMD Ryzen 9"))
              .upgrade_memory(64)
              .upgrade_hard_drive_capacity(8);
    
    println!("Gaming rig: {:#?}", gaming_rig);
}

// =============================================================================
// STUDY GUIDE SUMMARY
// =============================================================================

/*
KEY TAKEAWAYS FROM THE BUILDER PATTERN:

1. PROBLEM: Methods that return `()` cannot be chained, leading to repetitive code.

2. SOLUTION: Make methods return `&mut Self` to enable chaining.

3. MECHANISM: 
   - Change return type from implicit `()` to `-> &mut Self`
   - Return `self` as the last expression (without semicolon)

4. BENEFITS:
   - Improved readability and fluent API
   - Reduced code verbosity
   - Efficient in-place mutation without ownership moves

5. RUST-SPECIFIC ADVANTAGES:
   - `&mut self` allows modification while maintaining ownership
   - Automatic dereferencing makes chaining seamless
   - Memory efficient - no copying or moving of data

BEFORE (Non-chainable):
    computer.upgrade_cpu(String::from("M4 Max"));
    computer.upgrade_memory(128);
    computer.upgrade_hard_drive_capacity(4);

AFTER (Chainable):
    computer.upgrade_cpu(String::from("M4 Max"))
            .upgrade_memory(128)
            .upgrade_hard_drive_capacity(4);

The Builder Pattern is a design choice that makes sequential modifications
to a single instance both readable and concise - a hallmark of well-designed APIs.
*/
```
