```rust
// Demonstrating different types of structs, their instantiation, methods, and associated functions in Rust

// Named field struct
#[derive(Debug, Clone)]
struct MenuItem {
    name: String,
    price: f64,
}

// Tuple struct
#[derive(Debug)]
struct Point(i32, i32);

// Unit-like struct
#[derive(Debug)]
struct Marker;

// Implementation block for named field struct
impl MenuItem {
    // Associated function (constructor)
    fn new(name: String, price: f64) -> Self {
        MenuItem { name, price }
    }

    // Method with immutable reference
    fn get_name(&self) -> &String {
        &self.name
    }

    // Method with mutable reference
    fn update_price(&mut self, new_price: f64) {
        self.price = new_price;
    }

    // Method with ownership
    fn transfer_ownership(self) -> String {
        self.name
    }

    // Method with mutable ownership
    fn mutate_and_return(mut self) -> Self {
        self.price += 1.0;
        self
    }
}

// Implementation block for tuple struct
impl Point {
    // Associated function (constructor)
    fn new(x: i32, y: i32) -> Self {
        Point(x, y)
    }

    // Method to calculate distance from origin
    fn distance_from_origin(&self) -> f64 {
        ((self.0.pow(2) + self.1.pow(2)) as f64).sqrt()
    }
}

// Implementation block for unit-like struct
impl Marker {
    // Associated function (constructor)
    fn new() -> Self {
        Marker
    }

    // Method for unit-like struct
    fn identify(&self) -> &'static str {
        "I am a Marker struct!"
    }
}

fn main() {
    // Named field struct instantiation
    let name = String::from("Latte");
    let price = 4.5;
    let mut item1 = MenuItem { name, price }; // Using field init shorthand
    println!("Item1: {:?}", item1); // Outputs: MenuItem { name: "Latte", price: 4.5 }

    // Using struct update syntax
    let item2 = MenuItem {
        price: 5.0,
        ..item1.clone() // Clone to avoid moving ownership
    };
    println!("Item2: {:?}", item2); // Outputs: MenuItem { name: "Latte", price: 5.0 }

    // Using constructor
    let mut item3 = MenuItem::new(String::from("Cappuccino"), 4.0);
    println!("Item3: {:?}", item3); // Outputs: MenuItem { name: "Cappuccino", price: 4.0 }

    // Method calls
    println!("Name: {}", item3.get_name()); // Outputs: Cappuccino
    item3.update_price(4.75);
    println!("Updated item3: {:?}", item3); // Outputs: MenuItem { name: "Cappuccino", price: 4.75 }

    // Ownership transfer
    let name = item3.transfer_ownership();
    println!("Transferred name: {}", name); // Outputs: Cappuccino
    // println!("{:?}", item3); // ERROR: item3 moved

    // Mutable ownership
    let item4 = MenuItem::new(String::from("Espresso"), 3.5);
    let item4 = item4.mutate_and_return();
    println!("Mutated item4: {:?}", item4); // Outputs: MenuItem { name: "Espresso", price: 4.5 }

    // Tuple struct instantiation
    let point = Point::new(3, 4);
    println!("Point: {:?}", point); // Outputs: Point(3, 4)
    println!("Distance from origin: {}", point.distance_from_origin()); // Outputs: 5.0

    // Accessing tuple struct fields
    println!("X coordinate: {}", point.0); // Outputs: 3

    // Unit-like struct instantiation
    let marker = Marker::new();
    println!("Marker: {:?}", marker); // Outputs: Marker
    println!("Marker says: {}", marker.identify()); // Outputs: I am a Marker struct!

    // Demonstrating ownership with structs
    let item5 = MenuItem::new(String::from("Mocha"), 5.0);
    let item6 = item5; // Ownership moves to item6
    // println!("{:?}", item5); // ERROR: item5 moved
    println!("Item6: {:?}", item6); // Outputs: MenuItem { name: "Mocha", price: 5.0 }

    // Method invocation on reference
    let item7 = MenuItem::new(String::from("Flat White"), 4.25);
    let item_ref = &item7;
    println!("Name via reference: {}", item_ref.get_name()); // Outputs: Flat White (auto-dereferenced)
}
```
