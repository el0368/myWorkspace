### Part 1: Introduction to Methods and the `impl` Block

**Example**: Defining a method using an `impl` block compared to a plain function.

```rust
struct TaylorSwiftSong {
    title: String,
    duration: u32, // in seconds
}

// Method defined in impl block
impl TaylorSwiftSong {
    fn print_title(&self) {
        println!("Song title: {}", self.title);
    }
}

// Plain function (not a method)
fn print_song_title(song: &TaylorSwiftSong) {
    println!("Song title: {}", song.title);
}

fn main() {
    let song = TaylorSwiftSong {
        title: String::from("Love Story"),
        duration: 234,
    };
    song.print_title(); // Method call
    print_song_title(&song); // Function call
}
```

### Part 2: The `self` Parameter with Ownership

**Example**: Using `self` and `mut self` to take ownership of the struct instance.

```rust
struct TaylorSwiftSong {
    title: String,
    duration: u32,
}

impl TaylorSwiftSong {
    // Takes ownership, moves the instance
    fn consume_self(self) {
        println!("Consumed song: {}", self.title);
    }
    
    // Takes mutable ownership, moves and modifies
    fn consume_and_change(mut self) {
        self.title = format!("{} (Remix)", self.title);
        println!("Modified and consumed: {}", self.title);
    }
}

fn main() {
    let song = TaylorSwiftSong {
        title: String::from("Blank Space"),
        duration: 231,
    };
    song.consume_self();
    // println!("{:?}", song); // Error: song was moved
}
```

### Part 3: The `self` Parameter with References

**Example**: Using `&self` and `&mut self` for immutable and mutable references.

```rust
struct TaylorSwiftSong {
    title: String,
    duration: u32,
}

impl TaylorSwiftSong {
    // Immutable reference
    fn get_title(&self) -> &String {
        &self.title
    }
    
    // Mutable reference
    fn append_to_title(&mut self, suffix: &str) {
        self.title.push_str(suffix);
    }
}

fn main() {
    let mut song = TaylorSwiftSong {
        title: String::from("Shake It Off"),
        duration: 219,
    };
    println!("Title: {}", song.get_title());
    song.append_to_title(" (Live)");
    println!("Updated title: {}", song.get_title());
}
```

### Part 6: Summary of `self` Parameter Options

**Example**: Showing all four `self` parameter options in one `impl` block.

```rust
struct TaylorSwiftSong {
    title: String,
    duration: u32,
}

impl TaylorSwiftSong {
    // Immutable reference
    fn read_title(&self) {
        println!("Read: {}", self.title);
    }
    
    // Mutable reference
    fn update_title(&mut self, new_title: String) {
        self.title = new_title;
    }
    
    // Immutable ownership
    fn take_self(self) {
        println!("Took: {}", self.title);
    }
    
    // Mutable ownership
    fn take_and_change(mut self) {
        self.title = String::from("Changed");
        println!("Changed and took: {}", self.title);
    }
}

fn main() {
    let mut song = TaylorSwiftSong {
        title: String::from("Style"),
        duration: 230,
    };
    song.read_title();
    song.update_title(String::from("Style (Updated)"));
    song.read_title();
    song.take_self();
    // song.read_title(); // Error: song was moved
}
```
