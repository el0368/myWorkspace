### Part 2: Defining the Method with mut self

struct TaylorSwiftSong {
    title: String,
    duration: u32,
}

impl TaylorSwiftSong {
    // Mutable ownership with mut self
    fn change_title(mut self) {
        self.title = String::from("New Title");
        println!("Changed title to: {}", self.title);
    }
    
    // Verbose syntax with explicit type
    fn change_duration(mut self: TaylorSwiftSong) {
        self.duration = 300;
        println!("Changed duration to: {}", self.duration);
    }
    
    // Using Self alias
    fn change_both(mut self: Self) {
        self.title = String::from("Updated Title");

        println!("Updated title: {}, duration: {}", self.title, self.duration);
    }
}

    let song = TaylorSwiftSong {

    };
    song.change_title();
