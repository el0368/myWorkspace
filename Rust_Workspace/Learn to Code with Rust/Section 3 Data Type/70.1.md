Here are code examples for each concept from the section review:

### Integers (Signed and Unsigned)


```Rust
fn main() {
    // Signed integers (i family)
    let my_signed_int: i8 = -120;
    let another_signed_int: i32 = 1_000_000;

    // Unsigned integers (u family)
    let my_unsigned_int: u8 = 200;
    let another_unsigned_int: u64 = 1_000_000_000_000;

    println!("Signed: {}, Unsigned: {}", my_signed_int, my_unsigned_int);
    println!("Another signed: {}, Another unsigned: {}", another_signed_int, another_unsigned_int);
}
```

### Floats (f32 and f64)


```Rust
fn main() {
```

### Format Specifier (Float Precision)


```Rust
fn main() {
```

### Mathematical Operations


```Rust
fn main() {
    let a = 10;
    let b = 3;

    // Addition
    let sum = a + b;
    println!("{} + {} = {}", a, b, sum);

    // Subtraction
    let difference = a - b;
    println!("{} - {} = {}", a, b, difference);

    // Multiplication
    let product = a * b;
    println!("{} * {} = {}", a, b, product);

    // Division (integer division - floor division)
    let quotient_int = a / b;
    println!("{} / {} (integer division) = {}", a, b, quotient_int);

    // Division (floating point division)
    let c = 10.0;
    let d = 3.0;
    let quotient_float = c / d;
    println!("{} / {} (float division) = {}", c, d, quotient_float);

    // Modulo (remainder)
    let remainder = a % b;
    println!("{} % {} = {}", a, b, remainder);
}
```

### Augmented Assignment Operators


```Rust
fn main() {
    let mut x = 5;
    println!("Initial x: {}", x);

    x += 3; // x = x + 3
    println!("x after += 3: {}", x);

    x -= 2; // x = x - 2
    println!("x after -= 2: {}", x);

    x *= 4; // x = x * 4
    println!("x after *= 4: {}", x);

    x /= 2; // x = x / 2
    println!("x after /= 2: {}", x);

    x %= 3; // x = x % 3
    println!("x after %= 3: {}", x);
}
```

### Booleans and Comparison/Logical Operators


```Rust
fn main() {
    let is_active = true;
    let is_logged_in = false;

    println!("Is active: {}", is_active);
    println!("Is logged in: {}", is_logged_in);

    let num1 = 10;
    let num2 = 20;

    // Equality
    println!("{} == {}: {}", num1, num2, num1 == num2);

    // Inequality
    println!("{} != {}: {}", num1, num2, num1 != num2);

    // Greater than
    println!("{} > {}: {}", num1, num2, num1 > num2);

    // Less than
    println!("{} < {}: {}", num1, num2, num1 < num2);

    // Greater than or equal to
    println!("{} >= {}: {}", num1, num1, num1 >= num1);

    // Less than or equal to
    println!("{} <= {}: {}", num1, num2, num1 <= num2);

    // AND operator
    let condition1 = true;
    let condition2 = false;
    println!("{} && {}: {}", condition1, condition2, condition1 && condition2);

    // OR operator
    println!("{} || {}: {}", condition1, condition2, condition1 || condition2);
}
```

### Traits (Debug and Display)


```Rust
use std::fmt;

// A custom struct that doesn't implement Display by default
struct Point {
    x: i32,
    y: i32,
}

// Manually implement the Display trait for Point
impl fmt::Display for Point {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        write!(f, "Point({}, {})", self.x, self.y)
    }
}

// Automatically derive the Debug trait for another struct
#[derive(Debug)]
struct Rectangle {
    width: u32,
    height: u32,
}

fn main() {
    // Display trait (for user-friendly output)
    let my_number = 42;
    println!("Display of number: {}", my_number);

    let p = Point { x: 10, y: 20 };
    println!("Display of Point: {}", p);

    // Debug trait (for developer-friendly output)
    let numbers_array = [1, 2, 3];
    println!("Debug of array: {:?}", numbers_array);

    let my_tuple = ("hello", 123, true);
    println!("Debug of tuple: {:?}", my_tuple);

    let rect = Rectangle { width: 30, height: 50 };
    println!("Debug of Rectangle: {:?}", rect);

    // Pretty-printed Debug
    println!("Pretty-printed Debug of Rectangle: {:#?}", rect);
}
```

### Ranges


```Rust
fn main() {
    // Exclusive range (b is not included)
    for i in 1..5 {
        print!("{} ", i); // Prints: 1 2 3 4
    }
    println!();

    // Inclusive range (b is included)
    for i in 1..=5 {
        print!("{} ", i); // Prints: 1 2 3 4 5
    }
    println!();

    // Range of characters
    for c in 'a'..='d' {
        print!("{} ", c); // Prints: a b c d
    }
    println!();
}
```

### Generics


```Rust
// A generic function that works with any type T
fn print_anything<T>(value: T) {
    // To print a generic type, it often needs to implement the Debug trait
    // This example won't compile without `T` implementing `Debug`
    // However, for illustration, imagine `T` is something printable
    // println!("The value is: {}", value); // This would require `T: Display`
    // So we'll use Debug for a more general example
    println!("The value (debug format) is: {:?}", value);
}

// A generic struct
struct Box<T> {
    item: T,
}

fn main() {
    // Using the generic function with different types
    print_anything(123); // T is i32
    print_anything("hello"); // T is &str
    print_anything(true); // T is bool

    // Creating instances of the generic struct with different types
    let int_box = Box { item: 10 };
    let string_box = Box { item: "Rust" };
    let float_box = Box { item: 3.14 };

    // To print the Box, it also needs Debug derived
    #[derive(Debug)]
    struct BoxDebug<T> {
        item: T,
    }

    let int_box_debug = BoxDebug { item: 10 };
    let string_box_debug = BoxDebug { item: "Rust" };

    println!("Int Box: {:?}", int_box_debug);
    println!("String Box: {:?}", string_box_debug);
}
```
