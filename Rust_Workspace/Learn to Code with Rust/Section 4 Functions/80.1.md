Here are simple code examples for each concept from the section review:

### Function Definition and Invocation

Rust

```Rust
fn greet() {
    println!("Hello from a function!");
}

fn main() {
    greet(); // Invoking the function
}
```

### Parameters and Arguments

Rust

```Rust
fn say_hello_to(name: &str, age: i32) {
    println!("Hello, {}! You are {} years old.", name, age);
}

fn main() {
    say_hello_to("Alice", 30); // "Alice" and 30 are arguments
    say_hello_to("Bob", 25);
}
```

### Explicit Return Value

Rust

```Rust
fn add_numbers_explicit(a: i32, b: i32) -> i32 {
    let sum = a + b;
    return sum; // Explicit return with `return` keyword and semicolon
}

fn main() {
    let result = add_numbers_explicit(5, 3);
    println!("Explicit sum: {}", result);
}
```

### Implicit Return Value

Rust

```Rust
fn add_numbers_implicit(a: i32, b: i32) -> i32 {
    a + b // Implicit return (no `return` keyword, no semicolon)
}

fn main() {
    let result = add_numbers_implicit(10, 7);
    println!("Implicit sum: {}", result);
}
```

### Function Returning a Unit (No Return Value)

Rust

```Rust
fn do_nothing() {
    println!("This function does nothing useful, but it runs.");
    // Implicitly returns () (unit type)
}

fn main() {
    do_nothing();
}
```

### Arbitrary Block with Return Value

Rust

```Rust
fn calculate_block_value() {
    let x = 10;

    let y = {
        let z = x * 2;
        z + 5 // This is the implicit return value of the block
    }; // Semicolon here to assign the block's value to `y`

    println!("Value from block: {}", y);
}

fn main() {
    calculate_block_value();
}
```
