fn main() {
    // --- 1. The Purpose of References (Borrowing) ---
    // We create a String. `s1` is the owner.
    let s1 = String::from("hello");

    // We pass a reference `&s1` to the function. Ownership is NOT moved.
    let len = calculate_length(&s1);

    // `s1` is still valid here because we only borrowed it.
    println!("The length of '{}' is {}.", s1, len);
    println!("---");

    // --- 2. The Two Types of References ---

    // A. Immutable References (&T) - "Multiple Readers" are OK.
    let s2 = String::from("readers");
    let r1 = &s2;
    let r2 = &s2;
    println!("Multiple immutable references are fine: {} and {}", r1, r2);

    // Immutable references implement `Copy`. `r1` is copied to `r3`.
    let r3 = r1;
    println!("`r1` is still valid after copy: {} and {}", r1, r3);
    println!("---");

    // B. Mutable References (&mut T) - "One Writer" only.
    let mut s3 = String::from("writer");
    let mr1 = &mut s3;
    // let mr2 = &mut s3; // ERROR: Cannot create a second mutable borrow.
    println!("One mutable reference is fine: {}", mr1);

    // Mutable references do NOT implement `Copy`, they are MOVED.
    let mr2 = mr1; // `mr1` is moved to `mr2`.
    // println!("{}", mr1); // ERROR: `mr1` is no longer valid.
    println!("After move, only new mutable ref is valid: {}", mr2);
    println!("---");

    // --- 3. The Borrow Checker and Lifetimes (NLL) ---
    let mut s4 = String::from("lifetime");
    let mut_ref1 = &mut s4;
    mut_ref1.push_str(" check"); // `mut_ref1`'s lifetime ends after its last use.

    let immut_ref1 = &s4; // This is OK, because `mut_ref1` is no longer used.
    println!("s4 is now: '{}'", immut_ref1);
    println!("---");

    // --- 4. Preventing Dangling References ---
    // The `dangle` function below would cause a compile error because it tries
    // to return a reference to data that goes out of scope.
    // let dangling_reference = dangle();
    println!("Rust prevents dangling references.");
    println!("---");

    // --- 5. Borrowing from Collections ---
    // With `Copy` types (like i32), the value is copied.
    let copy_array = [1, 2, 3];
    let first_element_copy = copy_array[0];
    println!("The array `{:?}` is still valid after copy.", copy_array);
    println!("The copied element is {}", first_element_copy);

    // With non-`Copy` types (like String), you must borrow or clone.
    let move_array = [String::from("a"), String::from("b")];
    // let first_element_move = move_array[0]; // ERROR: Cannot move out of array.
    
    // Solution 1: Borrow the element.
    let first_element_borrow = &move_array[0];
    println!("Borrowed element: {}", first_element_borrow);
    
    // Solution 2: Clone the element.
    let first_element_clone = move_array[0].clone();
    println!("Cloned element: {}", first_element_clone);
    
    // The original array is still valid in both cases.
    println!("The `move_array` is still valid: {:?}", move_array);
}

// Function that takes an immutable reference (borrows).
fn calculate_length(s: &String) -> usize {
    s.len()
} // `s` goes out of scope, but the data it refers to is NOT dropped.
