// Example demonstrating Option and Result enums in Rust

fn main() {
    // Option enum examples
    let some_value: Option<i32> = Some(42);
    let none_value: Option<i32> = None;

    // Using unwrap on Option
    let unwrapped = some_value.unwrap();
    println!("Unwrapped Some: {}", unwrapped); // Outputs: 42
    // let bad_unwrap = none_value.unwrap(); // Would panic!

    // Using expect on Option
    let expected = some_value.expect("Expected a value!");
    println!("Expected Some: {}", expected); // Outputs: 42
    // let bad_expect = none_value.expect("No value found!"); // Would panic with custom message

    // Using unwrap_or on Option
    let safe_some = some_value.unwrap_or(0);
    let safe_none = none_value.unwrap_or(0);
    println!("Unwrap_or Some: {}", safe_some); // Outputs: 42
    println!("Unwrap_or None: {}", safe_none); // Outputs: 0

    // Match with Option
    match some_value {
        Some(value) => println!("Match Some: {}", value), // Outputs: 42
        None => println!("Match None"),
    }
    match none_value {
        Some(value) => println!("Match Some: {}", value),
        None => println!("Match None"), // Outputs: Match None
    }

    // If let with Option
    if let Some(value) = some_value {
        println!("If let Some: {}", value); // Outputs: 42
    }
    if let Some(value) = none_value {
        println!("If let Some: {}", value);
    } else {
        println!("If let None"); // Outputs: If let None
    }

    // Result enum examples
    let ok_result: Result<i32, &str> = Ok(100);
    let err_result: Result<i32, &str> = Err("Something went wrong");

    // Using unwrap on Result
    let ok_unwrapped = ok_result.unwrap();
    println!("Unwrapped Ok: {}", ok_unwrapped); // Outputs: 100
    // let bad_unwrap = err_result.unwrap(); // Would panic!

    // Using expect on Result
    let ok_expected = ok_result.expect("Expected a success!");
    println!("Expected Ok: {}", ok_expected); // Outputs: 100
    // let bad_expect = err_result.expect("Custom error message"); // Would panic with custom message

    // Using unwrap_or on Result
    let safe_ok = ok_result.unwrap_or(-1);
    let safe_err = err_result.unwrap_or(-1);
    println!("Unwrap_or Ok: {}", safe_ok); // Outputs: 100
    println!("Unwrap_or Err: {}", safe_err); // Outputs: -1

    // Match with Result
    match ok_result {
        Ok(value) => println!("Match Ok: {}", value), // Outputs: 100
        Err(e) => println!("Match Err: {}", e),
    }
    match err_result {
        Ok(value) => println!("Match Ok: {}", value),
        Err(e) => println!("Match Err: {}", e), // Outputs: Something went wrong
    }

    // If let with Result
    if let Ok(value) = ok_result {
        println!("If let Ok: {}", value); // Outputs: 100
    }
    if let Err(e) = err_result {
        println!("If let Err: {}", e); // Outputs: Something went wrong
    }

    // Debug trait with Result
    println!("Result debug: {:?}", ok_result); // Outputs: Ok(100)
    println!("Result debug: {:?}", err_result); // Outputs: Err("Something went wrong")

    // Using variants without enum prefix (Rust prelude)
    let direct_some = Some(10);
    let direct_none = None::<i32>;
    let direct_ok = Ok::<i32, &str>(50);
    let direct_err = Err::<i32, &str>("Failed");

    println!("Direct Some: {:?}", direct_some); // Outputs: Some(10)
    println!("Direct None: {:?}", direct_none); // Outputs: None
    println!("Direct Ok: {:?}", direct_ok); // Outputs: Ok(50)
    println!("Direct Err: {:?}", direct_err); // Outputs: Err("Failed")
}

// Example function returning Option
fn divide(a: i32, b: i32) -> Option<i32> {
    if b == 0 {
        None
    } else {
        Some(a / b)
    }
}

// Example function returning Result
fn checked_divide(a: i32, b: i32) -> Result<i32, &'static str> {
    if b == 0 {
        Err("Division by zero")
    } else {
        Ok(a / b)
    }
}
