// Comprehensive example demonstrating vectors in Rust, covering all details from the transcript

fn main() {
    // Creating a vector with Vec::new() and type annotation
    let mut vec1: Vec<i32> = Vec::new(); // Empty vector of i32
    vec1.push(1); // Infer type from push, but annotation provided
    println!("Vec1 after push: {:?}", vec1); // Outputs: [1]

    // Creating with turbofish syntax
    let mut vec2 = Vec::<String>::new();
    vec2.push(String::from("Hello"));
    println!("Vec2: {:?}", vec2); // Outputs: ["Hello"]

    // Infer type by pushing
    let mut vec3 = Vec::new();
    vec3.push(3.14); // Infers Vec<f64>
    println!("Vec3: {:?}", vec3); // Outputs: [3.14]

    // Using vec! macro
    let vec4 = vec![true, false, true]; // Vec<bool>
    println!("Vec4: {:?}", vec4); // Outputs: [true, false, true]

    // Mutation methods: push (append to end)
    let mut vec5: Vec<i32> = Vec::new();
    vec5.push(10);
    vec5.push(20);
    println!("Vec5 after push: {:?}", vec5); // Outputs: [10, 20]

    // Insert at specific index
    vec5.insert(0, 5); // Prepend to beginning
    vec5.insert(2, 15); // Insert in middle
    println!("Vec5 after insert: {:?}", vec5); // Outputs: [5, 10, 15, 20]

    // Pop: removes last, returns Option
    let popped = vec5.pop();
    println!("Popped: {:?}", popped); // Outputs: Some(20)
    println!("Vec5 after pop: {:?}", vec5); // Outputs: [5, 10, 15]
    let empty_pop: Option<i32> = Vec::<i32>::new().pop();
    println!("Pop from empty: {:?}", empty_pop); // Outputs: None

    // Remove by index, returns value (panics if invalid)
    let removed = vec5.remove(1);
    println!("Removed: {}", removed); // Outputs: 10
    println!("Vec5 after remove: {:?}", vec5); // Outputs: [5, 15]
    // vec5.remove(10); // Would panic: index out of bounds

    // Reading elements: [] with ownership rules
    let vec6 = vec![1, 2, 3]; // Copy type (i32)
    let elem_copy = vec6[1]; // Copies value
    println!("Elem copy: {}", elem_copy); // Outputs: 2
    println!("Vec6 still: {:?}", vec6); // Outputs: [1, 2, 3]

    let vec7 = vec![String::from("A"), String::from("B")]; // Non-Copy type
    // let elem_move = vec7[0]; // ERROR: Would move ownership out
    let elem_borrow = &vec7[0]; // Borrow reference
    println!("Elem borrow: {}", elem_borrow); // Outputs: A
    println!("Vec7 still: {:?}", vec7); // Outputs: ["A", "B"]

    // Using get: returns Option<&T>
    if let Some(value) = vec7.get(1) {
        println!("Get index 1: {}", value); // Outputs: B
    }
    let invalid_get = vec7.get(10);
    println!("Invalid get: {:?}", invalid_get); // Outputs: None

    // Borrowing a slice
    let slice = &vec7[0..1]; // &["A"]
    println!("Slice: {:?}", slice); // Outputs: ["A"]
    let full_slice = &vec7[..]; // Entire vector as slice
    println!("Full slice: {:?}", full_slice); // Outputs: ["A", "B"]
    let partial_slice = &vec7[1..]; // From index 1 to end
    println!("Partial slice: {:?}", partial_slice); // Outputs: ["B"]

    // Writing/replacing elements
    let mut vec8 = vec![100, 200, 300];
    vec8[1] = 250; // Overwrite
    println!("Vec8 after write: {:?}", vec8); // Outputs: [100, 250, 300]

    // Borrowing rules: One mutable ref at a time
    let mut vec9 = vec![1, 2, 3];
    let mut_ref = &mut vec9;
    mut_ref.push(4); // Mutable access
    println!("Mutable ref: {:?}", mut_ref); // Outputs: [1, 2, 3, 4]
    // let immut_ref = &vec9; // ERROR: Cannot borrow while mutable borrow exists

    // After mutable ref no longer used, can borrow again
    let immut_ref1 = &vec9;
    let immut_ref2 = &vec9; // Multiple immutable refs allowed
    println!("Immutable ref1: {:?}", immut_ref1); // Outputs: [1, 2, 3, 4]
    println!("Immutable ref2: {:?}", immut_ref2); // Outputs: [1, 2, 3, 4]
}
