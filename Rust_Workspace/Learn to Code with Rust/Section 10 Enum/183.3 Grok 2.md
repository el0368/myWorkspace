```rust
// Example 1: Basic Enum Declaration
// Demonstrates a simple enum with variants and no associated data
#[derive(Debug)]
enum Direction {
    Up,
    Down,
    Left,
    Right,
}

// Example 2: Enum with Different Types of Variants
// Shows tuple variants, struct variants, and empty variants
#[derive(Debug)]
enum Message {
    Quit,                        // Empty variant
    Move { x: i32, y: i32 },    // Struct variant
    Write(String),              // Tuple variant with one field
    ChangeColor(u8, u8, u8),    // Tuple variant with three fields
}

// Example 3: Ownership and Enums
// Enums do not implement Copy by default, so ownership rules apply
fn demonstrate_ownership() {
    let msg = Message::Write(String::from("Hello"));
    // msg is moved into process_message if not borrowed
    // To avoid moving, we can borrow with &msg
}

// Example 4: Tuple Variant Instantiation
fn create_tuple_variant() {
    let color = Message::ChangeColor(255, 128, 0);
    println!("{:?}", color); // Prints: ChangeColor(255, 128, 0)
}

// Example 5: Struct Variant Instantiation
fn create_struct_variant() {
    let movement = Message::Move { x: 10, y: 20 };
    println!("{:?}", movement); // Prints: Move { x: 10, y: 20 }
}

// Example 6: Empty Variant Instantiation
fn create_empty_variant() {
    let quit = Message::Quit;
    println!("{:?}", quit); // Prints: Quit
}

// Example 7: Pattern Matching with `match`
fn process_message(msg: &Message) {
    match msg {
        Message::Quit => println!("Quitting..."),
        Message::Move { x, y } => println!("Moving to ({}, {})", x, y),
        Message::Write(text) => println!("Writing: {}", text),
        Message::ChangeColor(r, g, b) => println!("Color: ({}, {}, {})", r, g, b),
    }
}

// Example 8: Pattern Matching with Specific Data
fn match_specific_data(msg: &Message) {
    match msg {
        Message::Write(text) if text == "Hello" => println!("Specific message: {}", text),
        Message::Write(text) => println!("Other message: {}", text),
        _ => println!("Not a Write message"),
    }
}

// Example 9: Methods on Enums
impl Message {
    fn is_quit(&self) -> bool {
        matches!(self, Message::Quit)
    }

    fn describe(&self) -> String {
        match self {
            Message::Quit => "Quit message".to_string(),
            Message::Move { x, y } => format!("Move to ({}, {})", x, y),
            Message::Write(text) => format!("Write: {}", text),
            Message::ChangeColor(r, g, b) => format!("Color: ({}, {}, {})", r, g, b),
        }
    }
}

// Example 10: `if let` Construct
fn check_write_message(msg: &Message) {
    if let Message::Write(text) = msg {
        println!("Text message: {}", text);
    } else {
        println!("Not a Write message");
    }
}

// Example 11: Combining Enums with Structs
#[derive(Debug)]
struct Point {
    x: i32,
    y: i32,
}

#[derive(Debug)]
enum Shape {
    Circle { center: Point, radius: f32 },
    Rectangle { top_left: Point, width: f32, height: f32 },
}

impl Shape {
    fn area(&self) -> f32 {
        match self {
            Shape::Circle { radius, .. } => 3.14159 * radius * radius,
            Shape::Rectangle { width, height, .. } => width * height,
        }
    }
}

// Main function to demonstrate all examples
fn main() {
    // Demonstrate Direction enum
    let up = Direction::Up;
    println!("Direction: {:?}", up);

    // Demonstrate Message enum with different variants
    let quit = Message::Quit;
    let move_msg = Message::Move { x: 5, y: 10 };
    let write = Message::Write(String::from("Hello"));
    let color = Message::ChangeColor(255, 0, 0);

    // Test pattern matching
    process_message(&quit);
    process_message(&move_msg);
    process_message(&write);
    process_message(&color);

    // Test specific data matching
    match_specific_data(&write);
    match_specific_data(&quit);

    // Test methods
    println!("Is quit? {}", quit.is_quit());
    println!("Description: {}", write.describe());

    // Test if let
    check_write_message(&write);
    check_write_message(&quit);

    // Test Shape enum with struct
    let point = Point { x: 0, y: 0 };
    let circle = Shape::Circle { center: point, radius: 5.0 };
    let rectangle = Shape::Rectangle { top_left: point, width: 3.0, height: 4.0 };
    println!("Circle area: {}", circle.area());
    println!("Rectangle area: {}", rectangle.area());
}
```
