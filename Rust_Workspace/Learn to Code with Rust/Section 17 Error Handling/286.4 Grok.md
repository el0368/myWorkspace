// Comprehensive example demonstrating error handling in Rust, covering all details from the transcript

use std::fs::File;
use std::io::{self, Read};
use std::process;

// Function that causes a panic (unrecoverable error)
fn cause_panic(index: usize) {
    let vec = vec![1, 2, 3];
    // Accessing invalid index will panic with backtrace
    println!("Invalid access: {}", vec[index]);
}

// Function to gracefully exit with an exit code
fn exit_program(code: i32) {
    println!("Exiting with code: {}", code);
    process::exit(code);
}

// Function to demonstrate standard error output
fn log_error(message: &str) {
    eprintln!("Error: {}", message); // Outputs to stderr
}

// Function returning Result for recoverable error
fn read_file_contents(path: &str) -> Result<String, io::Error> {
    let mut file = File::open(path)?; // ? operator propagates error
    let mut contents = String::new();
    file.read_to_string(&mut contents)?;
    Ok(contents)
}

// Function returning Option
fn find_number(numbers: &[i32], target: i32) -> Option<usize> {
    for (index, &num) in numbers.iter().enumerate() {
        if num == target {
            return Some(index);
        }
    }
    None
}

fn main() {
    // Demonstrating panic (unrecoverable error)
    // cause_panic(10); // Would panic with backtrace

    // Graceful exit
    if false { // Disabled to continue program
        exit_program(1); // Would exit with code 1 (error)
        // exit_program(0); // Would exit with code 0 (success)
    }

    // Standard error output
    log_error("Something went wrong!"); // Outputs to stderr

    // Handling Result with match
    let file_result = read_file_contents("nonexistent.txt");
    match file_result {
        Ok(contents) => println!("File contents: {}", contents),
        Err(e) => eprintln!("Failed to read file: {}", e), // Outputs to stderr
    }

    // Handling Result with if let
    if let Ok(contents) = read_file_contents("example.txt") {
        println!("File contents: {}", contents);
    } else {
        eprintln!("Could not read example.txt");
    }

    // Using ? operator to propagate errors
    fn process_file(path: &str) -> Result<String, io::Error> {
        let contents = read_file_contents(path)?; // Propagates error
        Ok(contents.to_uppercase())
    }

    match process_file("nonexistent.txt") {
        Ok(upper) => println!("Processed: {}", upper),
        Err(e) => eprintln!("Process error: {}", e),
    }

    // Method chaining with ? operator
    fn chain_operations(path: &str) -> Result<String, io::Error> {
        let contents = File::open(path)?.metadata()?.is_file().to_string();
        Ok(contents)
    }

    match chain_operations("example.txt") {
        Ok(result) => println!("Chained result: {}", result),
        Err(e) => eprintln!("Chain error: {}", e),
    }

    // Handling Option with match
    let numbers = vec![10, 20, 30];
    match find_number(&numbers, 20) {
        Some(index) => println!("Found 20 at index: {}", index), // Outputs: 1
        None => println!("Number not found"),
    }

    // Handling Option with if let
    if let Some(index) = find_number(&numbers, 40) {
        println!("Found 40 at index: {}", index);
    } else {
        println!("40 not found"); // Outputs: 40 not found
    }

    // Using ? operator with Option
    fn process_number(numbers: &[i32], target: i32) -> Option<usize> {
        let index = find_number(numbers, target)?;
        Some(index + 1) // Adjust index
    }

    match process_number(&numbers, 20) {
        Some(adjusted) => println!("Adjusted index: {}", adjusted), // Outputs: 2
        None => println!("No number found"),
    }
}
