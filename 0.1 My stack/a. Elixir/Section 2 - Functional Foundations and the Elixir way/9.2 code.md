defmodule PatternMatchingDemo do
  @moduledoc """
  Examples based on the "Elixir Pattern Matching: The Ultimate Shape Sorter" guide.
  """

  # --- 1. The Basics: It's Not Assignment! ---
  def basics_demo do
    # This is a match. Elixir binds 10 to x.
    x = 10
    IO.puts("1. Basics: x is bound to #{x}")

    # This succeeds because 10 matches the current value of x.
    10 = x
    IO.puts("   Match successful: 10 = x")

    # This will fail because 20 does not match 10.
    try do
      20 = x
    rescue
      _e in MatchError ->
        IO.puts("   Match failed: 20 = x (Raises MatchError as expected)")
    end
  end

  # --- 2. Tuples: The "Locked Box" Pattern ---
  def tuple_demo do
    # Simulating a common API response
    response = {:ok, "User data retrieved successfully"}

    # If the first element is :ok, bind the second element to 'message'
    {:ok, message} = response
    IO.puts("2. Tuples: Extracted message: '#{message}'")

    # Use the underscore (_) to ignore parts of the data
    {:error, _} = {:error, "Access Denied"}
    IO.puts("   Successfully ignored the error message using '_'")
  end

  # --- 3. Lists: The "Head and Tail" Trick ---
  def list_demo do
    numbers = [1, 2, 3, 4, 5]

    # Split the list into the first element and the rest
    [head | tail] = numbers
    IO.puts("3. Lists: Head is #{head}, Tail is #{inspect(tail)}")

    # Matching fixed lengths
    [a, b, c, d, e] = numbers
    IO.puts("   Fixed match: a=#{a}, b=#{b}, c=#{c}, d=#{d}, e=#{e}")
  end

  # --- 4. Maps: The "Pick and Choose" Match ---
  def map_demo do
    user = %{name: "Alice", age: 30, city: "London", job: "Engineer"}

    # Partial Matching: We only care about the :name key
    %{name: user_name} = user
    IO.puts("4. Maps: Found user name: #{user_name}")

    # Deep Extraction: Pulling data out of nested maps
    nested = %{status: %{code: 200, info: "OK"}}
    %{status: %{code: status_code}} = nested
    IO.puts("   Nested Extraction: HTTP Status Code = #{status_code}")
  end

  # --- 5. Function Heads: The Ultimate "If-Else" Killer ---
  
  # Clause 1: Matches exactly the atom :hello
  def greet(:hello), do: "Well, hi there!"

  # Clause 2: Matches a map that contains a :name key
  def greet(%{name: name}), do: "Hello, #{name}!"

  # Clause 3: Fallback for any other input
  def greet(_), do: "I'm not sure who you are."

  # --- 6. Guards: The "Bouncer" at the Door ---

  # Uses the 'when' keyword for additional logic checks
  def check_age(age) when age >= 18, do: "Access granted: Adult"
  def check_age(age) when age >= 13, do: "Access granted: Teenager"
  def check_age(_), do: "Access denied: Too young"

end

# --- Script Execution ---

IO.puts("=== Starting Pattern Matching Demo ===\n")

PatternMatchingDemo.basics_demo()
PatternMatchingDemo.tuple_demo()
PatternMatchingDemo.list_demo()
PatternMatchingDemo.map_demo()

IO.puts("\n5. Testing Function Heads:")
IO.puts(PatternMatchingDemo.greet(:hello))
IO.puts(PatternMatchingDemo.greet(%{name: "Alice"}))
IO.puts(PatternMatchingDemo.greet("Random String"))

IO.puts("\n6. Testing Guards:")
IO.puts(PatternMatchingDemo.check_age(25))
IO.puts(PatternMatchingDemo.check_age(15))
IO.puts(PatternMatchingDemo.check_age(10))

IO.puts("\n=== Demo Complete ===")
